#+title: X.Org

These are configuration snippets for the X server. No [[id:bbe0ff18-1b29-4f04-b2a6-5f3d7b003eb5::Makefile][Makefile]] rule installs them,
so we copy the ones that a machine needs by hand. The ~k0~ machine's own X
configuration is in [[id:82cbef73-61b9-43e7-9df6-7b2771000733::nixos/k0/xorg.conf][~doc/nix.org~]].

* Monitor layouts

#+name: dual-monitor-hybrid
#+header: :tangle xorg/10-dual-monitor-hybrid.conf
#+begin_src conf

    Identifier     "Layout0"

* Server flags

These set the timeouts for blanking the screen and for turning it off.

#+name: server-flags
#+header: :tangle xorg/10-server-flags.conf
#+begin_src conf
Section "ServerFlags"
    Option "BlankTime" "50"
    Option "StandbyTime" "60"
    Option "SuspendTime" "70"
    Option "OffTime" "80"
EndSection
#+end_src

which method is active.

* Input devices

These set the keyboard layout, disable mouse acceleration, and disable tapping on
touchpads.

#+name: keyboard
#+header: :tangle xorg/10-keyboard.conf
#+begin_src conf
Section "InputClass"
    Identifier "Keyboard Defaults"
    MatchIsKeyboard "yes"
    Option  "XkbLayout" "us"
EndSection
#+end_src

#+name: mouse
#+header: :tangle xorg/50-mouse.conf
#+begin_src conf
Section "InputClass"
	Identifier "My Mouse"
	MatchIsPointer "yes"
	# Disable all acceleration/deceleration
	Option "AccelerationProfile" "-1"
	Option "AccelerationScheme" "none"
EndSection
#+end_src

#+name: synaptics
#+header: :tangle xorg/50-synaptics.conf
#+begin_src conf
Section "InputClass"
        Identifier "touchpad catchall"
        Driver "synaptics"
        MatchIsTouchpad "on"
        MatchDevicePath "/dev/input/event*"
        Option "TapButton1" "0"
        Option "TapButton2" "0"
        Option "TapButton3" "0"
EndSection
#+end_src
